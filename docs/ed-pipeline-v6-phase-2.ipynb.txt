class EnhancedResultsNotifier:
    """
    Complete HL7v2 Results Processing with troponin delta detection.
    Source: ED_Pathway_Trainer_PoC_v2_fixed.ipynb ResultsNotifier class
    """
    def __init__(self):
        self.callbacks = []
        self.last_values = {}  # {patient_id: {test_code: (value, timestamp)}}
        self.delta_rules = {"TROPONIN": {"abs_ng_per_l": 5.0, "rel_pct": 20.0}}
        
    def on_notify(self, fn): 
        self.callbacks.append(fn)
        
    def _emit(self, pid, event, payload):
        for cb in self.callbacks: 
            try:
                cb(pid, event, payload)
            except Exception:
                pass
    
    @staticmethod
    def _split_segments(msg: str):
        """Handle both \\r and \\n line endings in HL7 messages."""
        norm = msg.replace('\r\n','\n').replace('\r','\n')
        return [s.split('|') for s in norm.strip().split('\n') if s]
    
    @staticmethod
    def _field(component: str, idx: int) -> str:
        parts = component.split('^')
        return parts[idx] if idx < len(parts) else ''
    
    @staticmethod
    def _parse_ts(ts: str):
        """Parse HL7 timestamp formats."""
        if not RUN_PIPELINE:
            return None
            
        for fmt in ('%Y%m%d%H%M%S','%Y%m%d%H%M','%Y%m%d'):
            try: 
                return pd.Timestamp.strptime(ts, fmt)
            except Exception: 
                pass
        return None
    
    def _get_pid(self, segs):
        for s in segs:
            if s[0]=='PID': 
                return s[3].split('^')[0] if len(s)>3 else ''
        return ''
    
    def handle_hl7(self, message: str):
        """
        Process HL7v2 ORU^R01 and MDM^T02 messages.
        Source: ED_Pathway_Trainer_PoC_v2_fixed.ipynb
        """
        if not RUN_PIPELINE:
            return []
            
        segs = self._split_segments(message)
        if not segs or segs[0][0] != 'MSH': 
            return []
            
        msg_type = segs[0][8] if len(segs[0])>8 else ''
        pid = self._get_pid(segs) or 'UNKNOWN'
        
        if 'ORU^R01' in msg_type: 
            return self._handle_oru(pid, segs)
        elif 'MDM^T02' in msg_type or ('ORU^R01' in msg_type and 
                                      any(s[0]=='OBX' and len(s)>2 and s[2] in ('TX','FT','ED') for s in segs)):
            return self._handle_report(pid, segs)
        return []
    
    def _handle_oru(self, pid, segs):
        """Process ORU^R01 lab results with delta detection."""
        obr_accession, obr_ts = None, None
        results = []
        
        for s in segs:
            if s[0]=='OBR':
                obr_accession = s[3] if len(s)>3 else None
                obr_ts = self._parse_ts(s[7]) if len(s)>7 else None
            elif s[0]=='OBX':
                id_comp = s[3] if len(s)>3 else ''
                code = self._field(id_comp,0) or self._field(id_comp,1) or 'UNKNOWN_TEST'
                value_raw = s[5] if len(s)>5 else ''
                units = s[6] if len(s)>6 else ''
                status = s[11] if len(s)>11 else ''
                ts = self._parse_ts(s[14]) if len(s)>14 else obr_ts
                
                try: 
                    value = float(value_raw)
                except Exception: 
                    value = None
                
                payload = {
                    'test_code': code.upper(),
                    'value_raw': value_raw,
                    'value': value,
                    'units': units,
                    'status': status,
                    'ts': ts,
                    'accession': obr_accession
                }
                
                self._emit(pid, 'lab_result_ready', payload)
                results.append(payload)
                
                if status and status.upper().startswith('C'): 
                    self._emit(pid, 'lab_result_critical', payload)
                    
                if value is not None: 
                    delta_result = self._maybe_emit_delta(pid, payload)
                    if delta_result:
                        results.append(delta_result)
        
        return results
    
    def _maybe_emit_delta(self, pid, payload):
        """
        Check for significant troponin delta based on age-adjusted thresholds.
        Source: Clinical Requirements - 14-51: 20%, >51: 50%
        """
        code, value = payload['test_code'], payload['value']
        ts = payload['ts'] or pd.Timestamp.utcnow()
        
        rules = self.delta_rules.get(code)
        if not rules:
            self.last_values.setdefault(pid,{})[code] = (value, ts)
            return None
            
        last = self.last_values.get(pid,{}).get(code)
        if last:
            prev, _ = last
            abs_delta = abs(value - prev)
            rel_pct = (abs_delta/prev*100.0) if prev else 0.0
            
            if (abs_delta >= rules.get('abs_ng_per_l', 1e9)) or (rel_pct >= rules.get('rel_pct', 1e9)):
                delta_payload = {
                    **payload,
                    'prev_value': prev,
                    'abs_delta': abs_delta,
                    'rel_pct': rel_pct,
                    'event_type': 'delta_positive'
                }
                self._emit(pid, 'lab_delta_positive', delta_payload)
                self.last_values.setdefault(pid,{})[code] = (value, ts)
                return delta_payload
        
        self.last_values.setdefault(pid,{})[code] = (value, ts)
        return None
    
    def _handle_report(self, pid, segs):
        """Process imaging and other text reports."""
        text_blocks, study_id, ts = [], None, None
        
        for s in segs:
            if s[0]=='OBR':
                study_id = s[3] if len(s)>3 else study_id
                ts = self._parse_ts(s[7]) if len(s)>7 else ts
            elif s[0]=='OBX' and len(s)>2 and s[2] in ('TX','FT','ED'):
                text_blocks.append(s[5] if len(s)>5 else '')
        
        if text_blocks:
            payload = {
                'study_id': study_id,
                'report_text': '\n'.join(text_blocks),
                'ts': ts
            }
            self._emit(pid, 'imaging_report_ready', payload)
            return [payload]
        return []

class EnhancedReminderEngine:
    """
    Timer-based clinical reminder system.
    Source: ED_Pathway_Trainer_PoC_v2_fixed.ipynb ReminderEngine class
    """
    def __init__(self, now_fn=None):
        self.now_fn = now_fn or (lambda: pd.Timestamp.utcnow())
        self.patients = {}  # {patient_id: {events: {}, timers: []}}
        self.callbacks = []
        
    def on_notify(self, fn): 
        self.callbacks.append(fn)
        
    def _emit(self, pid, key, payload):
        for cb in self.callbacks: 
            try:
                cb(pid, key, payload)
            except Exception:
                pass
    
    def ensure_patient(self, pid): 
        self.patients.setdefault(pid, {"events": {}, "timers": []})
    
    def ingest_event(self, pid, event_type, payload=None):
        """Record patient event and evaluate timers."""
        if not RUN_PIPELINE:
            return
            
        payload = payload or {}
        self.ensure_patient(pid)
        now = self.now_fn()
        
        self.patients[pid]["events"].setdefault(event_type, []).append({
            "t": now, 
            **payload
        })
        self._evaluate(pid)
    
    def add_timer(self, pid, key, due_in, condition):
        """Add a timer for patient with given condition."""
        if not RUN_PIPELINE:
            return
            
        self.ensure_patient(pid)
        due_at = self.now_fn() + due_in
        
        self.patients[pid]["timers"].append({
            "key": key,
            "due_at": due_at,
            "condition": condition,
            "fired": False
        })
    
    def _has_event(self, pid, event_type, where=None):
        """Check if patient has specific event matching criteria."""
        evs = self.patients.get(pid,{}).get("events",{}).get(event_type,[])
        if where is None: 
            return bool(evs)
            
        return any(all(e.get(k)==v for k,v in where.items()) for e in evs)
    
    def _evaluate(self, pid):
        """Check if any timers are due for this patient."""
        now = self.now_fn()
        
        for timer in self.patients[pid]["timers"]:
            if timer["fired"]: 
                continue
                
            if now >= timer["due_at"]:
                cond = timer["condition"]
                ok = True
                
                if "require_event" in cond: 
                    ok = self._has_event(pid, cond["require_event"], cond.get("where"))
                    
                if not ok: 
                    timer["fired"] = True
                    self._emit(pid, timer["key"], {
                        "due_at": timer["due_at"], 
                        "condition": cond
                    })
    
    def apply_lab_protocol(self, pid, protocol):
        """
        Apply lab protocol with timing reminders.
        Source: Clinical Requirements - Serial troponin protocols
        """
        if not RUN_PIPELINE:
            return
            
        self.ensure_patient(pid)
        
        # Initial order reminder
        self.add_timer(
            pid, 
            f"{protocol['test_name']}_initial_order_due",
            pd.Timedelta(minutes=protocol.get("initial_order_due_min", 10)),
            {"require_event": "order_placed", "where": {"test": protocol["test_name"]}}
        )
        
        self.patients[pid].setdefault("lab_protocols", {})[protocol["test_name"]] = protocol
    
    def on_sample_collected(self, pid, test_name):
        """Handle sample collection and schedule repeats."""
        if not RUN_PIPELINE:
            return
            
        proto = self.patients.get(pid,{}).get("lab_protocols",{}).get(test_name)
        if not proto: 
            return
            
        for mins in proto.get("repeat_schedule_min", []):
            self.add_timer(
                pid, 
                f"{test_name}_repeat_due_{mins}m",
                pd.Timedelta(minutes=mins),
                {"require_event": "sample_collected", "where": {"test": test_name, "offset_min": mins}}
            )
